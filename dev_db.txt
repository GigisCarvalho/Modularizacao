-- BANCO DE DADOS FICTÍCIO PARA TREINAMENTO DE CRUD
-- MySQL

CREATE DATABASE IF NOT EXISTS crud_usuarios
    CHARACTER SET utf8mb4
    COLLATE utf8mb4_unicode_ci;

USE crud_usuarios;

DROP TABLE IF EXISTS usuarios;

CREATE TABLE usuarios (
    id BIGINT UNSIGNED NOT NULL AUTO_INCREMENT,
    nome VARCHAR(150) NOT NULL,
    email VARCHAR(255) NOT NULL,
    telefone VARCHAR(20) NULL,
    data_nascimento DATE NULL,
    cidade VARCHAR(100) NULL,
    estado CHAR(2) NULL,
    cargo VARCHAR(100) NULL,
    ativo BOOLEAN NOT NULL DEFAULT TRUE,
    criado_em TIMESTAMP NOT NULL DEFAULT CURRENT_TIMESTAMP,
    atualizado_em TIMESTAMP NOT NULL DEFAULT CURRENT_TIMESTAMP
        ON UPDATE CURRENT_TIMESTAMP,

    PRIMARY KEY (id),
    CONSTRAINT uk_usuarios_email UNIQUE (email)
);

-- CONSULTAS PARA VALIDAR OS DADOS

-- Listar todos os usuários
SELECT *
FROM usuarios
ORDER BY id;

-- Consultar usuário pelo ID
SELECT *
FROM usuarios
WHERE id = 1;

-- Buscar usuários que contenham parte do nome
SELECT *
FROM usuarios
WHERE nome LIKE '%Ana%'
ORDER BY nome;

-- Busca por nome sem diferenciar letras maiúsculas e minúsculas
SELECT *
FROM usuarios
WHERE LOWER(nome) LIKE LOWER('%ana%')
ORDER BY nome;

-- Listar somente usuários ativos
SELECT *
FROM usuarios
WHERE ativo = TRUE
ORDER BY nome;

-- Conferir a quantidade de registros
SELECT COUNT(*) AS total_usuarios
FROM usuarios;
